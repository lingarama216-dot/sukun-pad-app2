#include #include #include #include // ===== MOTOR PINS ===== const int motorHead = 25; // Head motor relay const int motorBack = 26; // Back motor relay const int motorLegs = 27; // Legs motor relay // ===== BLE SETUP ===== #define SERVICE_UUID "4fafc201-1fb5-459e-8fcc-c5c9c331914b" #define CHARACTERISTIC_UUID "beb5483e-36e1-4688-b7f5-ea07361b26d8" bool deviceConnected = false; int currentIntensity = 50; String currentZone = "NONE"; String currentMode = "KNEAD"; class MyServerCallbacks: public BLEServerCallbacks { void onConnect(BLEServer* pServer) { deviceConnected = true; Serial.println("Device Connected"); }; void onDisconnect(BLEServer* pServer) { deviceConnected = false; stopAllMotors(); // Disconnect hote hi sab band Serial.println("Device Disconnected"); } }; class MyCallbacks: public BLECharacteristicCallbacks { void onWrite(BLECharacteristic *pCharacteristic) { std::string value = pCharacteristic->getValue(); if (value.length() > 0) { String command = String(value.c_str()); Serial.println("Command: " + command); handleCommand(command); } } }; // ===== MOTOR CONTROL FUNCTIONS ===== void stopAllMotors() { digitalWrite(motorHead, LOW); digitalWrite(motorBack, LOW); digitalWrite(motorLegs, LOW); currentZone = "NONE"; Serial.println("All Motors STOP"); } void runMotor(int pin, int intensity) { // Intensity 0-100 ko PWM 0-255 me convert int pwmValue = map(intensity, 0, 100, 0, 255); analogWrite(pin, pwmValue); } void handleCommand(String cmd) { // Format: ZONE|MODE|INTENSITY|TIME // Example: HEAD|KNEAD|70|15 if(cmd == "STOP") { stopAllMotors(); return; } // Zone alag karo if(cmd.indexOf("HEAD") >= 0) { stopAllMotors(); currentZone = "HEAD"; runMotor(motorHead, currentIntensity); } else if(cmd.indexOf("BACK") >= 0) { stopAllMotors(); currentZone = "BACK"; runMotor(motorBack, currentIntensity); } else if(cmd.indexOf("LEGS") >= 0) { stopAllMotors(); currentZone = "LEGS"; runMotor(motorLegs, currentIntensity); } // Intensity nikalna if(cmd.indexOf("|") >= 0) { int firstBar = cmd.indexOf("|"); int secondBar = cmd.indexOf("|", firstBar + 1); int thirdBar = cmd.indexOf("|", secondBar + 1); currentMode = cmd.substring(firstBar + 1, secondBar); currentIntensity = cmd.substring(secondBar + 1, thirdBar).toInt(); Serial.print("Mode: "); Serial.print(currentMode); Serial.print(" Intensity: "); Serial.println(currentIntensity); } } void setup() { Serial.begin(115200); // Motor pins ko output banao pinMode(motorHead, OUTPUT); pinMode(motorBack, OUTPUT); pinMode(motorLegs, OUTPUT); stopAllMotors(); // BLE Start BLEDevice::init("SUKOON-PAD"); BLEServer *pServer = BLEDevice::createServer(); pServer->setCallbacks(new MyServerCallbacks()); BLEService *pService = pServer->createService(SERVICE_UUID); BLECharacteristic *pCharacteristic = pService->createCharacteristic( CHARACTERISTIC_UUID, BLECharacteristic::PROPERTY_WRITE ); pCharacteristic->setCallbacks(new MyCallbacks()); pService->start(); BLEAdvertising *pAdvertising = BLEDevice::getAdvertising(); pAdvertising->addServiceUUID(SERVICE_UUID); pAdvertising->start(); Serial.println("SUKOON-PAD READY. Waiting for connection..."); } void loop() { // Yahan timer ka code daal sakte hain baad me delay(1000); }
